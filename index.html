        function verificarAnio() {

            const tipo = tipoSelect.value;

            const anio = anioAutoInput.value;

            calcularMesesMora();

            if (tipo && anio && anio.length === 4) {

                if (tarifario[tipo].datos[anio]) {

                    // Año existe en la tabla

                    divCategoria.classList.remove('d-none');

                    divMontoManual.classList.add('d-none');

                    catSelect.required = true;

                    montoBaseManualInput.required = false;

                } else {

                    // Año NO existe (muy viejo o futuro)

                    divCategoria.classList.add('d-none');

                    divMontoManual.classList.remove('d-none');

                    catSelect.required = false;

                    montoBaseManualInput.required = true;

                }

            } else {

                divCategoria.classList.add('d-none');

                divMontoManual.classList.add('d-none');

                catSelect.required = false;

                montoBaseManualInput.required = false;

            }

        }

        function calcularMesesMora() {

            const anio = parseInt(anioAutoInput.value);

            if (!anio || anio < 1900) return;

            const dPago = parseInt(diaActualInput.value) || 1;

            const mPago = parseInt(mesActualInput.value) || 1;

            const aPago = parseInt(anioActualInput.value) || today.getFullYear();

            const fechaVencimiento = new Date(anio, 0, 30); // 30 de Enero

            const fechaPago = new Date(aPago, mPago - 1, dPago);

            if (fechaPago <= fechaVencimiento) {

                mesesAtrasoManual.value = 0;

            } else {

                let meses = (fechaPago.getFullYear() - fechaVencimiento.getFullYear()) * 12;

                meses -= fechaVencimiento.getMonth();

                meses += fechaPago.getMonth();

               

                if (fechaPago.getDate() <= 30 && meses > 0) {

                    meses--;

                }

               

                mesesAtrasoManual.value = meses > 0 ? meses : 0;

            }

        }

        diaActualInput.addEventListener('input', calcularMesesMora);

        mesActualInput.addEventListener('input', calcularMesesMora);

        anioActualInput.addEventListener('input', calcularMesesMora);

        // Submit form and show results

        form.addEventListener('submit', function(e) {

            e.preventDefault();

           

            const tipo = tipoSelect.value;

            const anio = anioAutoInput.value;

            const mesesAtraso = parseInt(mesesAtrasoManual.value) || 0;

            const derechoOficina = parseFloat(derechoOficinaInput.value) || 0;

            let montoBase = 0;

            if (tarifario[tipo].datos[anio]) {

                const catIndex = catSelect.value;

                if (!catIndex) {

                    alert('Por favor seleccioná la categoría del vehículo.');

                    return;

                }

                montoBase = tarifario[tipo].datos[anio][catIndex];

            } else {

                montoBase = parseFloat(montoBaseManualInput.value);

                if (!montoBase || montoBase <= 0) {

                    alert('Por favor ingresá un monto base válido para este vehículo antiguo.');

                    return;

                }

            }

            const recargoMonto = montoBase * 0.01 * mesesAtraso;

            const total = montoBase + recargoMonto + derechoOficina;

            // UI Update

            document.getElementById('resBase').innerText = formatter.format(montoBase);

            document.getElementById('resMeses').innerText = mesesAtraso;

            document.getElementById('resRecargo').innerText = formatter.format(recargoMonto);

            document.getElementById('resOficina').innerText = formatter.format(derechoOficina);

            document.getElementById('resTotal').innerText = formatter.format(total);

        });

    </script>
